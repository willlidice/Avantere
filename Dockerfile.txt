# ============================================
# DOCKERFILE - AVANTERE
# Receita para construir o app no servidor
# ============================================

# Etapa 1: Imagem base (Node.js leve)
FROM node:18-alpine AS base

# --------------------------------------------
# Etapa 2: Instalar dependências
# --------------------------------------------
FROM base AS deps
RUN apk add --no-cache libc6-compat
WORKDIR /app

# Copiar arquivos de configuração de pacotes
COPY package.json package-lock.json* ./
COPY prisma ./prisma/

# Instalar todas as dependências
RUN npm ci

# Gerar o cliente do Prisma
RUN npx prisma generate

# --------------------------------------------
# Etapa 3: Construir a aplicação
# --------------------------------------------
FROM base AS builder
WORKDIR /app

# Copiar dependências da etapa anterior
COPY --from=deps /app/node_modules ./node_modules

# Copiar todo o código do projeto
COPY . .

# Configurar para produção
ENV NEXT_TELEMETRY_DISABLED 1
ENV NODE_ENV production

# Construir o app (npm run build)
RUN npm run build

# --------------------------------------------
# Etapa 4: Imagem final (menor possível)
# --------------------------------------------
FROM base AS runner
WORKDIR /app

# Configurar ambiente
ENV NODE_ENV production
ENV NEXT_TELEMETRY_DISABLED 1

# Criar usuário seguro (não usar root)
RUN addgroup --system --gid 1001 nodejs
RUN adduser --system --uid 1001 nextjs

# Copiar arquivos públicos
COPY --from=builder /app/public ./public
COPY --from=builder /app/prisma ./prisma

# Copiar o build do Next.js
COPY --from=builder --chown=nextjs:nodejs /app/.next/standalone ./
COPY --from=builder --chown=nextjs:nodejs /app/.next/static ./.next/static

# Usar o usuário seguro
USER nextjs

# Expor a porta 3000
EXPOSE 3000

# Configurar hostname
ENV PORT 3000
ENV HOSTNAME "0.0.0.0"

# Comando para iniciar o app
CMD ["node", "server.js"]
